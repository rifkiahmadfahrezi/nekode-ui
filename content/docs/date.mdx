title: Date Utilities
description: ISO date formatting and parsing helpers — yyyy-MM-dd, ISO strings, and UI display formatting.
---

Thin wrappers around the two conversions every form/table needs: `Date` → `yyyy-MM-dd` for API payloads, and an ISO string → `Date` for display, plus a handful of common checks (validity, same-day, past/future, relative time). All the date-picker fields in this registry (`DatePickerField`, `DateRangePickerField`, `DateMultiPickerField`) work with `Date` objects — these helpers sit at the boundary where your API or database uses strings instead.

Two implementations, same function names and behavior — pick one:

- **`date`** — built on `date-fns`. Use `formatStr`/`format`-style tokens (`"PPP"`, `"yyyy/MM/dd"`, ...).
- **`date-vanilla`** — zero dependencies, built on native `Date` and `Intl`. Formatting takes `Intl.DateTimeFormatOptions` instead of format tokens.

```ts title="lib/date.ts"
import {
  format,
  formatDistanceToNow,
  isBefore,
  isSameDay,
  isValid,
  parseISO,
} from "date-fns";

/** Format a `Date` as `yyyy-MM-dd` (e.g. for API payloads, `<input type="date">`). */
export function toISODateString(date: Date): string {
  return format(date, "yyyy-MM-dd");
}

/** Parse an ISO 8601 string — `"2026-09-19"` or `"2026-09-19T10:30:00Z"` — into a `Date`. */
export function parseISODate(value: string): Date {
  return parseISO(value);
}

/** Format a `Date` for display in the UI. Defaults to a long, human-readable format. */
export function formatDateDisplay(date: Date, formatStr = "PPP"): string {
  return format(date, formatStr);
}

/** Check whether a `Date` is a valid date (not `Invalid Date`). */
export function isValidDate(date: Date): boolean {
  return isValid(date);
}

/** Check whether an ISO 8601 string is valid and can be parsed into a `Date`. */
export function isValidISODate(value: string): boolean {
  return isValid(parseISO(value));
}

/** Format a `Date` as a time string (e.g. `"10:30 AM"`) for UI display. */
export function formatTimeDisplay(date: Date, formatStr = "p"): string {
  return format(date, formatStr);
}

/** Format a `Date` as a relative time string (e.g. `"2 days ago"`, `"in 3 hours"`). */
export function formatRelativeDate(date: Date): string {
  return formatDistanceToNow(date, { addSuffix: true });
}

/** Check whether two `Date`s fall on the same calendar day. */
export function isSameCalendarDay(a: Date, b: Date): boolean {
  return isSameDay(a, b);
}

/** Check whether a `Date` is before the current time. */
export function isPastDate(date: Date): boolean {
  return isBefore(date, new Date());
}
```

```ts title="lib/date-vanilla.ts"
interface FormatDisplayOptions {
  options?: Intl.DateTimeFormatOptions;
  locale?: string;
}

const RELATIVE_UNITS: [Intl.RelativeTimeFormatUnit, number][] = [
  ["year", 1000 * 60 * 60 * 24 * 365],
  ["month", 1000 * 60 * 60 * 24 * 30],
  ["day", 1000 * 60 * 60 * 24],
  ["hour", 1000 * 60 * 60],
  ["minute", 1000 * 60],
  ["second", 1000],
];

/** Format a `Date` as `yyyy-MM-dd` (e.g. for API payloads, `<input type="date">`). */
export function toISODateString(date: Date): string {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");
  return `${year}-${month}-${day}`;
}

/** Parse an ISO 8601 string — `"2026-09-19"` or `"2026-09-19T10:30:00Z"` — into a `Date`. */
export function parseISODate(value: string): Date {
  return new Date(value);
}

/**
 * Format a `Date` for display in the UI using `Intl.DateTimeFormat`.
 * Defaults to a long, human-readable date. Pass `options` to customize (e.g. `{ dateStyle: "short" }`).
 */
export function formatDateDisplay(
  date: Date,
  { options = { dateStyle: "long" }, locale }: FormatDisplayOptions = {},
): string {
  return new Intl.DateTimeFormat(locale, options).format(date);
}

/** Check whether a `Date` is a valid date (not `Invalid Date`). */
export function isValidDate(date: Date): boolean {
  return date instanceof Date && !Number.isNaN(date.getTime());
}

/** Check whether an ISO 8601 string is valid and can be parsed into a `Date`. */
export function isValidISODate(value: string): boolean {
  return isValidDate(new Date(value));
}

/** Format a `Date` as a time string (e.g. `"10:30 AM"`) for UI display. */
export function formatTimeDisplay(
  date: Date,
  { options = { timeStyle: "short" }, locale }: FormatDisplayOptions = {},
): string {
  return new Intl.DateTimeFormat(locale, options).format(date);
}

/** Format a `Date` as a relative time string (e.g. `"2 days ago"`, `"in 3 hours"`). */
export function formatRelativeDate(date: Date, locale?: string): string {
  const diffMs = date.getTime() - Date.now();
  const rtf = new Intl.RelativeTimeFormat(locale, { numeric: "auto" });

  for (const [unit, ms] of RELATIVE_UNITS) {
    if (Math.abs(diffMs) >= ms || unit === "second") {
      return rtf.format(Math.round(diffMs / ms), unit);
    }
  }

  return rtf.format(0, "second");
}

/** Check whether two `Date`s fall on the same calendar day. */
export function isSameCalendarDay(a: Date, b: Date): boolean {
  return (
    a.getFullYear() === b.getFullYear() &&
    a.getMonth() === b.getMonth() &&
    a.getDate() === b.getDate()
  );
}

/** Check whether a `Date` is before the current time. */
export function isPastDate(date: Date): boolean {
  return date.getTime() < Date.now();
}
```

## Usage

### `date` (date-fns)

```ts
import {
  toISODateString,
  parseISODate,
  formatDateDisplay,
  formatTimeDisplay,
  formatRelativeDate,
  isValidDate,
  isValidISODate,
  isSameCalendarDay,
  isPastDate,
} from "@/lib/date";

toISODateString(new Date(2026, 8, 19)); // "2026-09-19"

parseISODate("2026-09-19"); // Date
parseISODate("2026-09-19T10:30:00Z"); // Date

formatDateDisplay(new Date(2026, 8, 19)); // "September 19th, 2026"
formatDateDisplay(new Date(2026, 8, 19), "yyyy/MM/dd"); // "2026/09/19"

formatTimeDisplay(new Date(2026, 8, 19, 10, 30)); // "10:30 AM"

formatRelativeDate(new Date(Date.now() - 2 * 24 * 60 * 60 * 1000)); // "2 days ago"

isValidDate(new Date("not a date")); // false
isValidISODate("2026-09-19"); // true

isSameCalendarDay(new Date(2026, 8, 19, 9), new Date(2026, 8, 19, 22)); // true

isPastDate(new Date(2020, 0, 1)); // true
```

### `date-vanilla` (no dependencies)

Same functions, same names — `formatDateDisplay`/`formatTimeDisplay` take an `{ options?, locale? }` object (`options` is `Intl.DateTimeFormatOptions`) instead of a format string.

```ts
import {
  toISODateString,
  parseISODate,
  formatDateDisplay,
  formatTimeDisplay,
  formatRelativeDate,
  isValidDate,
  isValidISODate,
  isSameCalendarDay,
  isPastDate,
} from "@/lib/date-vanilla";

formatDateDisplay(new Date(2026, 8, 19)); // "September 19, 2026"
formatDateDisplay(new Date(2026, 8, 19), { options: { dateStyle: "short" } }); // "9/19/26"

formatTimeDisplay(new Date(2026, 8, 19, 10, 30)); // "10:30 AM"

formatRelativeDate(new Date(Date.now() - 2 * 24 * 60 * 60 * 1000)); // "2 days ago"
```

### With a date-picker field

```tsx
<DatePickerField
  label="Start date"
  value={value ? parseISODate(value) : undefined}
  onValueChange={(date) => setValue(date ? toISODateString(date) : "")}
/>
```

## API Reference

Same signatures in both implementations except `formatDateDisplay` and `formatTimeDisplay`, noted below.

| Function | Signature (`date`) | Signature (`date-vanilla`) | Description |
|---|---|---|---|
| `toISODateString` | `(date: Date) => string` | same | Formats a `Date` as `yyyy-MM-dd`. |
| `parseISODate` | `(value: string) => Date` | same | Parses an ISO 8601 string — date-only or with time — into a `Date`. |
| `formatDateDisplay` | `(date: Date, formatStr?: string) => string` — `date-fns` format string, default `"PPP"` | `(date: Date, opts?: { options?: Intl.DateTimeFormatOptions; locale?: string }) => string` — default `{ dateStyle: "long" }` | Formats a `Date` for UI display. |
| `formatTimeDisplay` | `(date: Date, formatStr?: string) => string` — default `"p"` | `(date: Date, opts?: { options?: Intl.DateTimeFormatOptions; locale?: string }) => string` — default `{ timeStyle: "short" }` | Formats a `Date` as a time string, e.g. `"10:30 AM"`. |
| `formatRelativeDate` | `(date: Date) => string` | `(date: Date, locale?: string) => string` | Formats a `Date` relative to now, e.g. `"2 days ago"`, `"in 3 hours"`. |
| `isValidDate` | `(date: Date) => boolean` | same | Checks whether a `Date` is valid (not `Invalid Date`). |
| `isValidISODate` | `(value: string) => boolean` | same | Checks whether an ISO 8601 string can be parsed into a valid `Date`. |
| `isSameCalendarDay` | `(a: Date, b: Date) => boolean` | same | Checks whether two `Date`s fall on the same calendar day. |
| `isPastDate` | `(date: Date) => boolean` | same | Checks whether a `Date` is before the current time. |
